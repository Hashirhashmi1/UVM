// INTERFACE
interface counter_if(input logic clk);

    logic rst;
    logic en;
    logic [3:0] count;

endinterface

// DUT
module counter (
    input  logic       clk,
    input  logic       rst,
    input  logic       en,
    output logic [3:0] count
);

    always_ff @(posedge clk) begin

        if (rst)
            count <= 4'd0;

        else if (en)
            count <= count + 1;

    end

endmodule
